## PRIMITIVE DATA TYPES
these are often inherited from C

void :: Void :: 0 byte :: holds nothing

int  :: Integral :: 4 byte :: holds a whole number

char :: Character :: 1 byte :: holds a letter

short :: Integral :: 2 bytes :: also holds a whole number, but its range is much smaller

long :: Integral :: 8/4 bytes :: also holds a whole number but its range is much bigger

double :: Floating point :: 8 bytes :: holds a fraction number

float :: Floating point :: 4 bytes :: also holds a fraction number

bool :: Boolean :: 1 bytes :: usually holds only true(1) or false(0)


## UNSIGNED INT VS SIGNED INT:

SIGNED INT: int op;

a signed integer is a integer that allows use for both positive and negative numbers

UNSIGNED INT: unsigned int op;

a unsigned integer is a integer that only allows use for positive numbers
